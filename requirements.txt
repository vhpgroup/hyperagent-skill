# Python dependencies for the Hyperagent document skills bundle.
#   pip install -r requirements.txt
#
# Requires Python 3.10+. No version pins: these are all stable, wide-compatibility
# libraries, and pinning would fight your existing environment. Pin them yourself
# if you need reproducible builds.

# --- pdf ---
pypdf                 # read/merge/split/rotate/encrypt
pdfplumber            # text + table extraction with coordinates
pdf2image             # render pages to images (needs poppler on PATH)
Pillow                # image handling for the form-filling workflow
reportlab             # create PDFs from scratch

# --- docx / pptx / xlsx (shared OOXML pipeline) ---
lxml                  # XML parsing for unpack/pack
defusedxml            # hardened XML parsing

# --- xlsx ---
openpyxl              # read/write .xlsx
pandas                # tabular analysis (used throughout the xlsx docs)

# --- trích text từ Office sang markdown ---
# Tài liệu docx/pptx gọi `python -m markitdown <file>` ở bước đọc nội dung.
markitdown            # Office -> markdown

# --- optional, only if you want OCR on scanned PDFs ---
# pytesseract         # also requires the tesseract binary

# --- research ---
trafilatura          # bóc nội dung chính khỏi HTML, tốt nhất cho bài viết
readability-lxml     # dự phòng khi trafilatura không ăn
ddgs                 # backend DuckDuckGo (không cần key)
# Exa dùng urllib của stdlib, không cần thư viện riêng

# --- browser ---
playwright           # sau khi cài: python -m playwright install chromium

# --- hsmt-analyzer ---
python-docx          # đọc bảng yêu cầu kỹ thuật trong DOCX
PyMuPDF              # trích text và render PDF dự phòng
libarchive-c         # giải nén file đính kèm ZIP/RAR/7z

# --- interactive ---
